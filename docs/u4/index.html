Clique em qualquer rota para ver a resposta real da API. Compare com o código da U3 — o Express elimina o if/else e organiza tudo em rotas declarativas.

Endpoints disponíveis

GET /api/v1/veiculos Lista todos os veículos
GET /api/v1/veiculos/:placa Busca veículo pela placa
GET /api/v1/motoristas Lista todos os motoristas
GET /api/v1/oficinas Lista todas as oficinas

Como o Express organiza isso

// app.js — Front Controller
const express = require('express');
const app = express();
app.use(express.json());

app.use('/api/v1/veiculos', require('./routes/veiculoRoutes'));
app.use('/api/v1/motoristas', require('./routes/motoristaRoutes'));
app.use('/api/v1/oficinas', require('./routes/oficinaRoutes'));

app.listen(3000);
// routes/veiculoRoutes.js
const router = require('express').Router();
const ctrl = require('../controllers/veiculoController');

router.get('/', ctrl.listar);
router.get('/:placa', ctrl.buscarPorPlaca);
router.post('/', ctrl.criar);
router.put('/:placa', ctrl.atualizar);
router.delete('/:placa', ctrl.remover);

module.exports = router;
