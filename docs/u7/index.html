Conceito — conexão com mysql2

Pool de conexões — não uma conexão por vez

Em vez de abrir e fechar uma conexão por requisição, usamos um pool — um conjunto de conexões reutilizáveis. Mais eficiente, mais robusto.

middlewares/db.js const mysql = require('mysql2/promise');

const db = mysql.createPool({
  host: process.env.DB_HOST,
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: 'frota_db',
  connectionLimit: 10 // máximo de conexões simultâneas
});

module.exports = db; // compartilhado por todos os models

O CRUD completo com mysql2

veiculoModel.js // busca todos os veículos ordenados
const listar = async () => {
  const [rows] = await db.execute(
    'SELECT * FROM VEICULO ORDER BY placa'
  );
  return rows; // array de objetos
};
veiculoModel.js // ? são parâmetros — protegem contra SQL injection
const criar = async ({ placa, modelo, ano, km_atual, status }) => {
  const [result] = await db.execute(
    'INSERT INTO VEICULO (placa, modelo, ano, km_atual, status)'
    + ' VALUES (?, ?, ?, ?, ?)',
    [placa, modelo, ano, km_atual, status]
  );
  return result.affectedRows; // 1 se inserido
};
veiculoModel.js const atualizar = async (placa, { modelo, ano, km_atual, status }) => {
  const [result] = await db.execute(
    'UPDATE VEICULO SET modelo=?, ano=?, km_atual=?, status=?'
    + ' WHERE placa=?',
    [modelo, ano, km_atual, status, placa]
  );
  return result.affectedRows; // 0 se não encontrou
};
veiculoModel.js const remover = async (placa) => {
  const [result] = await db.execute(
    'DELETE FROM VEICULO WHERE placa = ?',
    [placa]
  );
  return result.affectedRows;
};
🚨 SQL Injection: nunca concatene valores do usuário diretamente na query. Use sempre ? com mysql2 — ele escapa os valores automaticamente.
